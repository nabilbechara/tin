package main

import "anvil"
import "argo"

type Message struct {
	text str max 200
}

fn hello(q anvil.Req, w mut anvil.Out) ! {
	let reply = try within 200ms {
		try greet(q.PathParam("name"))
	}
	w.Json()
	argo.Put(mut w.Body, Message{text: reply})
}

fn main() ! {
	let r = anvil.NewRouter()
	r.Get(`/hello/{name}`, hello)
	try r.Serve(":8080")
}
